## 创建模型

完成的代码如下：

```Swift
// Landmark.swift
import Foundation
import SwiftUI
import CoreLocation

struct Landmark: Hashable, Codable {
	var id: Int
	var name: String
	var park: String
	var state: String
	var description: String

	private var imageName: String
	var image: Image {
		Image(imageName)
	}

	private var coordinates: Coordinates
	var locationCoordinate: CLLocationCoordinate2D {
		CLLocationCoordinate2D(
			latitude: coordinates.latitude,
			longitude: coordinates.longitude
		)
	}

	struct Coordinates: Hashable, Codable {
		var latitude: Double
		var longitude: Double
	}
}
```

### 通过 json 数据创建对象

```Swift
// ModelData.swift
import Foundation

var landmarks: [Landmark] = load("landmarkData.json")

func load<T: Decodable>(_ filename: String) > T {
	let data: Data

	guard let file = Bundle.main.url(forResource: filename, withExtension: nil)
	else {
		fatalError("Couldn't find \(filename) in main bundle.")
	}

	do {
		data = try Data(contentsOf: file)
	} catch {
		fatalError("Couldn't load \(filename) from main bundle:\n\(error)")
	}

	do {
		let decoder = JSONDecoder()
		return try decoder.decode(T.self, from data)
	} catch {
		fatalError("Couldn't parse \(filename) as \(T.self):\n\(error)")
	}
}
```

## 创建row视图&自定义row预览

完成的代码如下：

```Swift
// LandmarkRow.swift
import SwiftUI

struct LandmarkRow: View {
	var landmark: Landmark

	var body: some View {
		HStack {
			landmark.image
				.resizable()
				.frame(width: 50, heigth: 50)
			Text(landmark.name)

			Spacer()
		}
	}
}

#Preview {
	LandmarkRow(landmark: landmarks[0])
}
```

`#Preview`代码块在一个文件中可以出现多次，后果是画布上也会有多个预览且预览的名字默认是#Preview在代码中的行数

可以为`#Preview`指定名字，如下：

```Swift
#Preview("Turtle Rock") {
	LandmarkRow(landmark: landmarks[0])
}

#Preview("Salmon") {
	LandmarkRow(landmark: landmarks[2])
}
```

可以用`Group`把多个预览组合在一起：

```Swift
#Preview {
	Group {
		LandmarkRow(landmark: landmarks[0])
		LandmarkRow(landmark: landmarks[1])
	}
}
```

## 创建列表

适用 List 类型来展示一个列表。列表的元素可以是静态的，也可以是动态生成的。甚至可以混合静态和动态生成的视图。

```Swift
// LandmarkList.swift
import SwiftUI

struct LandmarkList: View {
	var body: some View {
		List {
			LandmarkRow(landmark: landmarks[0])
			LandmarkRow(landmark: landmarks[1])
		}
	}
}

#Preview {
	LandmarkList()
}
```

## 使列表动态

```Swift
// LandmarkList.swift
import SwiftUI

struct LandmarkList: View {
	var body: some View {
		List(landmarks, id: \.id) { landmark in
			LandmarkRow(landmark: landmark)
		}
	}
}

#Preview {
	LandmarkList()
}
```

列表与 identifiable 数据一起工作，有两种方式可以让我们的数据变成 identifiable：

* 创建列表时传入唯一键的 key path
* 让数据类型遵循 Identifiable 协议（有id属性）

## 在列表与详情之间配置导航

把列表嵌入`NavigationSplitView`，并把每个 row 嵌入`NavigationLink`中就可以配置到目标页面的跳转。

```Swift
// LandmarkDetail.swift 略
```

完成后的 LandmarkList.swift

```Swift
import SwiftUI

struct LandmarkList: View {
	var body: some View {
		NavigationSplitView {
			List(landmarks) { landmark in
				NavigationLink {
					// 向子页面传递数据
					LandmarkDetail(landmark: landmark)
				} label: {
					LandmarkRow(landmark: landmark)
				}
			}
			.navigationTitle("Landmarks")
		} detail: {
			Text("Select a Landmark")
		}
	}
}
```

NavigationSplitView 的第二个参数是一个展位视图，在iPhone上不需要，在iPad上需要。

修改后的 MapView.swift

```Swift
import SwiftUI
import MapKit


struct MapView: View {
    var coordinate: CLLocationCoordinate2D


    var body: some View {
        Map(position: .constant(.region(region)))
    }


    private var region: MKCoordinateRegion {
        MKCoordinateRegion(
            center: coordinate,
            span: MKCoordinateSpan(latitudeDelta: 0.2, longitudeDelta: 0.2)
        )
    }
}


#Preview {
    MapView(coordinate: CLLocationCoordinate2D(latitude: 34.011_286, longitude: -116.166_868))
}
```

注意创建 Map 的地方，参数是一个双向绑定的 position。因为我们不希望在操作地图时修改 position 的值，所以这里使用了`.constant()`修饰。
