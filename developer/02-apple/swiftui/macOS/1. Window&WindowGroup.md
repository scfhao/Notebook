默认的WindowGroup支持多个macOS窗口。如果你的应用只需要一个窗口，请改用Window。

```swift
import SwiftUI

@main
struct ImageGeneratorApp: App {
    var body: some Scene {
        WindowGroup {
            ContentView()
        }
    }
}
```

```swift
import SwiftUI

@main
struct ImageGeneratorApp: App {
    var body: some Scene {
        Window("ImageGenerator", id: "main") {
            ContentView()
        }
    }
}
```
